\needspace{6cm}
\item \needspace{5cm}
A rectangle and a trapezium are shown below. \\
\begin{center}
\begin{tikzpicture}[scale=0.6]
\begin{scope}[shift={(-3,0)}]
\draw[thick] (0,0) rectangle (6,3);
\node[below] at (3,0) {$8\text{ cm}$};
\node[left] at (0,1.5) {$x\text{ cm}$};
\end{scope}
\begin{scope}[shift={(8,0)}]
\draw[thick] (0,0) -- (6,0) -- (4.5,4) -- (1.5,4) -- cycle;
\draw[dashed] (1.5,4) -- (1.5,0);
\draw (1.5,0) rectangle (1.8,0.3);
\node[below] at (3,0) {$8\text{ cm}$};
\node[above] at (3,4) {$4\text{ cm}$};
\node[left] at (1.5,2) {$(x+1)\text{ cm}$};
\end{scope}
\end{tikzpicture}
\end{center}
The area of the rectangle is equal to the area of the trapezium. \\
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Find the value of $x$.
\examanswerslot{3cm}{0.25}{3}
\item Calculate the perimeter of the rectangle.
\examanswerslot{2cm}{0.25}{1}
\end{enumerate}
\vspace{1cm}
\needspace{6cm}
\item \needspace{5cm}
The diagram shows a composite L-shaped polygon. \\
All internal angles are right angles. \\
\begin{center}
\begin{tikzpicture}

\end{center}
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Calculate the perimeter of the polygon.
\examanswerslot{2cm}{0.25}{1}
\item Calculate the total area of the polygon.
\examanswerslot{2cm}{0.25}{3}
\end{enumerate}
\vspace{1cm}
\needspace{6cm}
\item \needspace{5cm}
A rectangle is shown below. All dimensions are in centimetres. \\
\begin{center}
\begin{tikzpicture}
\draw [thick] (0,0) rectangle (4.5,2.5);
\node at (2.25,-0.3) {$(2x - 1)$};

\end{tikzpicture}
\end{center}
The exact area of the rectangle is $18$ cm$^2$. \\
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Show that the area equation simplifies to:
\par\vspace{0.1cm}\hspace*{0.6cm}$2x^2 + 3x - 20 = 0$\par\vspace{0.15cm}
\examanswerslot{3.5cm}{0}{2}
\item Solve the equation to find the value of $x$.
\examanswerslot{2cm}{0.25}{3}
\item Calculate the exact perimeter of the rectangle.
\examanswerslot{2cm}{0.25}{2}
\end{enumerate}
\vspace{1cm}
\needspace{6cm}
\item \needspace{5cm}
The points $P(-1, 2)$, $Q(3, 2)$, and $R(5, 6)$ are three vertices of a parallelogram $PQRS$ drawn on a coordinate grid. \\
\begin{center}
\begin{tikzpicture}[scale=0.8]
\draw[->, gray] (-2,0) -- (6,0) node[right] {$x$};
\draw[->, gray] (0,-1) -- (0,7) node[above] {$y$};
\draw[thick] (-1,2) -- (3,2) -- (5,6) -- (1,6) -- cycle;
\node[below left] at (-1,2) {$P(-1,2)$};
\node[below right] at (3,2) {$Q(3,2)$};
\node[above right] at (5,6) {$R(5,6)$};
\node[above left] at (1,6) {$S$};
\filldraw (-1,2) circle (1.5pt);
\filldraw (3,2) circle (1.5pt);
\filldraw (5,6) circle (1.5pt);
\filldraw (1,6) circle (1.5pt);
\end{tikzpicture}
\end{center}
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Write down the coordinates of the fourth vertex $S$.
\examanswerslot{2cm}{0.25}{1}
\item Calculate the area of the parallelogram $PQRS$.
\examanswerslot{2cm}{0.25}{2}
\item Find the exact length of the diagonal $PR$. Give your answer in the form $a\sqrt{b}$ where $a$ and $b$ are prime numbers.
\examanswerslot{2cm}{0.25}{3}
\end{enumerate}

\needspace{6cm}
\item \needspace{5cm}
The diagram shows a trapezium. \\
\begin{center}
\begin{tikzpicture}
\draw[thick] (0,0) -- (7,0) -- (5.5,2.5) -- (2,2.5) -- cycle;
\draw[dashed] (2,2.5) -- (2,0);

\node at (3.5,-0.3) {$14$ cm};
\node at (3.75,2.8) {$8$ cm};
\node at (1.5,1.25) {$5$ cm};
\end{tikzpicture}
\end{center}
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Calculate the area of the trapezium.
\examanswerslot{2cm}{0.25}{2}
\item A distinct triangle has the same area as the trapezium.
The triangle has a base of $11$ cm and a perpendicular height of $h$ cm. \\
\par\vspace{0.1cm}\hspace*{0.6cm}$\text{Area} = \dfrac{1}{2} \times \text{base} \times \text{height}$\par\vspace{0.15cm}
Show that the height, $h$, of the triangle is $10$ cm. \\
\examanswerslot{3.5cm}{0}{2}
\end{enumerate}

\needspace{6cm}
\item \needspace{5cm}
A sailing competition takes place around a triangular course $ABC$. The distance from mark $A$ to mark $B$ is 5 km, and the distance from $B$ to $C$ is 12 km. The angle at $B$ is exactly $90^\circ$. \\
\begin{center}
\begin{tikzpicture}
\draw[thick] (0,0) -- (4.8,0) -- (0,2) -- cycle;
\draw (0,0) rectangle (0.2,0.2);
\node[below] at (2.4,0) {12 km};
\node[left] at (0,1) {5 km};
\node[below left] at (0,0) {$B$};
\node[below right] at (4.8,0) {$C$};
\node[above left] at (0,2) {$A$};
\end{tikzpicture}
\end{center}
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Calculate the total perimeter of the sailing course.
\examanswerslot{2cm}{0.25}{3}
\item A leading boat completes the entire course at a constant average speed of 10 km/h.
\begin{enumerate}[label=\textbf{(\roman*)}, leftmargin=0.6cm]
\item Calculate the time taken for the boat to finish, giving your answer in hours and minutes.
\examanswerslot{2cm}{0.25}{2}
\item Next year, the organizers plan to extend the course boundaries to form a mathematically similar triangle whose enclosed area is 44\% larger than the current course. Calculate the perimeter of this new enlarged course.
\examanswerslot{2cm}{0.25}{3}
\end{enumerate}
\end{enumerate}

\item \needspace{5cm}

\begin{center}

\examanswerslot{2cm}{0.25}{2}

\item \needspace{5cm}
The parallel sides of a trapezium are in the ratio $3 : 5$. \\
\begin{center}
\begin{tikzpicture}
\draw [thick] (1.5,2.5) -- (4.5,2.5) -- (6,0) -- (0,0) -- cycle;
\draw [dashed] (1.5,2.5) -- (1.5,0);
\draw (1.5,0) rectangle (1.7,0.2);
\node at (2.1,1.25) {$8$ cm};
\end{tikzpicture}
\end{center}
The perpendicular height of the trapezium is $8$ cm. \\
The area of the trapezium is $96$ cm$^2$. \\
\begin{enumerate}[label=\textbf{(\alph*)}, leftmargin=*]
\item Calculate the lengths of the two parallel sides.
\examanswerslot{2cm}{0.25}{3}
\item The trapezium is isosceles. Calculate the exact perimeter of the trapezium. Give your answer in the form $p + q\sqrt{r}$ where $p, q$, and $r$ are integers.
\examanswerslot{2cm}{0.25}{4}
\end{enumerate}
\end{enumerate}
